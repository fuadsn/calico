cmake_minimum_required(VERSION 3.22)
project(calico_coach LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
include(FetchContent)
set(LLAMA_ARCHIVE "https://codeload.github.com/ggml-org/llama.cpp/tar.gz/3057bb66c86c46d5781e50e85462a760ba7d1feb")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../../../build/llama-source.tar.gz")
    set(LLAMA_ARCHIVE "${CMAKE_CURRENT_LIST_DIR}/../../../../build/llama-source.tar.gz")
endif()
FetchContent_Declare(llama URL "${LLAMA_ARCHIVE}"
    URL_HASH SHA256=624ecdc13c247c87a2bd315d365542a24639510b4e40d1b51deae9b55634e5cc)
FetchContent_MakeAvailable(llama)
add_library(calico_coach SHARED coach_jni.cpp)
target_link_libraries(calico_coach PRIVATE llama android log)
# NDK 29 also makes the dependency objects compatible with 16 KB Android pages.
target_link_options(calico_coach PRIVATE "-Wl,-z,max-page-size=16384")
